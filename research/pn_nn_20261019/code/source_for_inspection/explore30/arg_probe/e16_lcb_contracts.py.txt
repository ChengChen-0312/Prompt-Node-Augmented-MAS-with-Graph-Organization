class ContractError(ValueError):
    """Raised when an E16 value does not match the frozen contract."""

    "efficient solution. Output the final solution in a single ```python code block."
)

TEMPERATURE = 0.2

SEED_PARAMETER = None

PROMPT_MODES = ("QUESTION_ONLY", "WITH_STARTER")
ACTIVE_PROMPT_MODE = "WITH_STARTER"

def validate_prompt_mode(value: object) -> str:
    if type(value) is not str or value not in PROMPT_MODES:
        raise ContractError("prompt mode is unresolved or invalid")
    return value


def render_user_prompt(question_content: str, starter_code: str) -> str:
    """Render the exact frozen WITH_STARTER user message without normalization."""

    if type(question_content) is not str:
        raise ContractError("question content must be a string")
    if type(starter_code) is not str:
        raise ContractError("starter code must be a string")
    if starter_code == "":
        return question_content
    return question_content + "\n\n```python\n" + starter_code + "\n```"


@dataclass(frozen=True, slots=True)
class FrozenPrompt:
    """An arm-neutral exact prompt frozen once for reuse by A1 and A2."""

    mode: str
    system_text: str
    user_text: str

    def __post_init__(self) -> None:
        if self.mode != ACTIVE_PROMPT_MODE:
            raise ContractError("frozen prompt mode differs from active prompt mode")
        if self.system_text != SYSTEM_PROMPT:
            raise ContractError("frozen prompt system text differs")
        if type(self.user_text) is not str:
            raise ContractError("frozen prompt user text is invalid")
        try:
            self.system_text.encode("utf-8")
            self.user_text.encode("utf-8")
        except UnicodeEncodeError as error:
            raise ContractError("frozen prompt is not valid UTF-8 text") from error

    @property
    def system_bytes(self) -> bytes:
        return self.system_text.encode("utf-8")

    @property
    def user_bytes(self) -> bytes:
        return self.user_text.encode("utf-8")

    def for_arm(self, arm: str) -> "FrozenPrompt":
        if arm not in ("A1", "A2"):
            raise ContractError("prompt arm is invalid")
        return self


def freeze_prompt(question_content: str, starter_code: str) -> FrozenPrompt:
    return FrozenPrompt(
        mode=ACTIVE_PROMPT_MODE,
        system_text=SYSTEM_PROMPT,
        user_text=render_user_prompt(question_content, starter_code),
    )


def canonical_json_bytes(value: Any) -> bytes:
    """Return canonical UTF-8 JSON without a trailing newline."""

    try:
        return json.dumps(
            value,
            ensure_ascii=False,
            sort_keys=True,
            separators=(",", ":"),
            allow_nan=False,
        ).encode("utf-8")
    except (TypeError, ValueError) as error:
        raise ValueError("value is not canonical JSON") from error


def sha256_bytes(value: bytes) -> str:
    if type(value) is not bytes:
        raise TypeError("sha256 input must be bytes")
    return hashlib.sha256(value).hexdigest()


def canonical_sha256(value: Any) -> str:
    return sha256_bytes(canonical_json_bytes(value))


def validate_exact_fields(
    value: Mapping[str, Any], expected: Sequence[str], label: str
) -> None:
    if not isinstance(value, Mapping):
        raise ContractError(f"{label} must be an object")
    expected_set = set(expected)
    observed_set = set(value)
    missing = sorted(expected_set - observed_set)
    extra = sorted(observed_set - expected_set)
    if missing or extra:
        details: list[str] = []
        if missing:
            details.append(f"missing={','.join(missing)}")
        if extra:
            details.append(f"extra={','.join(extra)}")
        raise ContractError(f"{label} fields differ: {' '.join(details)}")


def validate_trace_row(value: Mapping[str, Any]) -> None:
    validate_exact_fields(value, TRACE_COLUMNS, "trace row")
    if value["arm"] not in ("A1", "A2"):
        raise ContractError("trace arm is invalid")
    if type(value["qid"]) is not str or not value["qid"]:
        raise ContractError("trace qid is invalid")
    if value["difficulty"] not in ("easy", "medium", "hard"):
        raise ContractError("trace difficulty is invalid")
    if type(value["date"]) is not str or not value["date"]:
        raise ContractError("trace date is invalid")
    for field in ("sample_idx", "prompt_chars", "completion_tokens"):
        if type(value[field]) is not int or value[field] < 0:
            raise ContractError(f"trace {field} is invalid")
    if type(value["finish_reason"]) is not str:
        raise ContractError("trace finish_reason is invalid")
    for field in ("extract_ok", "public_pass"):
        if type(value[field]) is not bool:
            raise ContractError(f"trace {field} is invalid")
    if value["official_pass"] is not None and type(value["official_pass"]) is not bool:
        raise ContractError("trace official_pass is invalid")
    if value["fail_mode"] not in FAIL_MODES:
        raise ContractError("trace fail_mode is invalid")
    if type(value["wall_time"]) not in (int, float) or value["wall_time"] < 0:
        raise ContractError("trace wall_time is invalid")


def _fsync_directory(path: Path) -> None:
    descriptor = os.open(path, os.O_RDONLY)
    try:
        os.fsync(descriptor)
    finally:
        os.close(descriptor)


def atomic_write_bytes(path: Path, data: bytes, *, replace: bool = False) -> None:
    """Atomically write exact bytes as 0600; no-clobber unless explicitly replaced."""

    if type(data) is not bytes:
        raise TypeError("atomic payload must be bytes")
    destination = Path(path)
    destination.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
    os.chmod(destination.parent, 0o700)
    descriptor, temporary_name = tempfile.mkstemp(
        prefix=f".{destination.name}.", suffix=".tmp", dir=destination.parent
    )
    temporary = Path(temporary_name)
    try:
        os.fchmod(descriptor, 0o600)
        with os.fdopen(descriptor, "wb") as stream:
            stream.write(data)
            stream.flush()
            os.fsync(stream.fileno())
        if replace:
            os.replace(temporary, destination)
        else:
            os.link(temporary, destination, follow_symlinks=False)
            temporary.unlink()
        os.chmod(destination, 0o600)
        observed = destination.stat(follow_symlinks=False)
        if not stat.S_ISREG(observed.st_mode) or stat.S_IMODE(observed.st_mode) != 0o600:
            raise ContractError("atomic destination is not an exact 0600 regular file")
        _fsync_directory(destination.parent)
    finally:
        try:
            os.close(descriptor)
        except OSError:
            pass
        try:
            temporary.unlink()
        except FileNotFoundError:
            pass


def write_canonical_json(path: Path, value: Any, *, replace: bool = False) -> str:
    payload = canonical_json_bytes(value)
    atomic_write_bytes(path, payload + b"\n", replace=replace)
    return sha256_bytes(payload)
