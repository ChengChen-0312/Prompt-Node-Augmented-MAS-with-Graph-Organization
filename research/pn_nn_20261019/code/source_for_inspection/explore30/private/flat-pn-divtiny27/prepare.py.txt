python prepare.py            -> materialize (no provider, no credential, no execution, no hidden read)
python prepare.py --freeze   -> write prereg + sha256 after checks.py passed
"""

DIVERSITY_INSTRUCTION = (
    'Earlier attempts at this problem took the approaches listed below. '
    'Produce a solution whose approach differs from all of them — not only in implementation, but in the algorithm it uses or in how it reads the problem statement. '
    'Briefly plan and state how your approach differs, then output the final solution in a single ```python code block.')


def digest(path): return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def dump(path, value): Path(path).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')


def plan_text(content):
    """Text before the first non-empty python fence (the program extract_python_code returns), capped."""
    start = next((m.start() for m in _PYTHON_FENCE.finditer(content) if m.group('code').strip()), None)
    text = (content if start is None else content[:start]).strip()
    return {'text': text[:PLAN_CHAR_LIMIT], 'truncated': len(text) > PLAN_CHAR_LIMIT, 'had_program': start is not None}


def solver_messages(task):
    return [{'role': 'system', 'content': SYSTEM_PROMPT},
            {'role': 'user', 'content': render_user_prompt(str(task['question_content']), str(task['starter_code']))}]


def control_payload(task):
    """The frozen E16 A1 solver call, byte-identical to the archived i.i.d. samples; reads nothing."""
    return {'model': MODEL, 'temperature': TEMPERATURE, 'max_tokens': MAX_TOKENS, 'messages': solver_messages(task)}


def arm_order(index):
    """Which arm sends first for this problem; alternating removes a systematic within-problem order effect."""
    return ('control', 'diversity') if index % 2 == 0 else ('diversity', 'control')


def diversity_payload(task, plans):
    body = DIVERSITY_INSTRUCTION + ''.join(f'\n\nPrevious approach {i}:\n{p}' for i, p in enumerate(plans, 1))
    return {'model': MODEL, 'temperature': TEMPERATURE, 'max_tokens': MAX_TOKENS,
            'messages': solver_messages(task) + [{'role': 'user', 'content': body}]}


def select():
    report = json.loads(AGREE_REPORT.read_text())
    rows = [p for p in report['per_problem_archive_only'] if p['group'] == 'dead' and p['n_fail'] >= 2 and p['small_output_space']]
    return sorted(rows, key=lambda p: p['qid'])


def materialize():
    if (STAGE / 'materialization.json').exists(): raise RuntimeError('ALREADY_MATERIALIZED')
    rows = select(); qids = [p['qid'] for p in rows]
    assert len(qids) == 27 and sum(p['S_distinct'] == 1 for p in rows) == 7
    tasks = {json.loads(l)['question_id']: json.loads(l) for l in (E21 / 'NOT_PAPER_EVIDENCE-e21p2-public-tasks.jsonl').open()}
    pool = defaultdict(list)
    for r in csv.DictReader(POOL.open()): pool[r['qid']].append(r)
    codes = {json.loads(l)['candidate_id']: json.loads(l)['code'] for l in (RUN / 'NOT_PAPER_EVIDENCE-e21p2-scoring-candidates.jsonl').open()}
    raw_by_ordinal = {int(Path(f).name.split('-')[0]): Path(f) for f in glob.glob(str(RUN / 'NOT_PAPER_EVIDENCE-e21p2-raw-responses' / '*.json'))}
    import tiktoken
    enc = tiktoken.encoding_for_model('gpt-4o')
    problems, estimate = [], Decimal(0)
    for index, p in enumerate(rows):
        q = p['qid']; task = tasks[q]
        initial = sorted((r for r in pool[q] if r['source'] == 'INITIAL'), key=lambda r: int(r['sample_idx']))
        assert [int(r['sample_idx']) for r in initial] == [0, 1, 2] and all(r['public_pass'] == 'False' for r in initial)
        archived_passes = sum(r['official_pass'] == 'True' for r in pool[q])
        assert archived_passes == 0, 'DEAD_PROBLEM_MUST_HAVE_NO_ARCHIVED_OFFICIAL_PASS'
        refs, plans = [], []
        for r in initial:
            path = raw_by_ordinal[int(r['ledger_ordinal'])]
            raw = json.loads(path.read_text()); content = raw['response']['choices'][0]['message']['content']
            # codes[...] is None when the archived response contained no fenced program (an extract failure in the archive)
            if raw['qid'] != q or raw['response'].get('model') != MODEL or extract_python_code(content) != codes[r['candidate_id']]:
                raise RuntimeError('INITIAL_RAW_RESPONSE_DOES_NOT_REPRODUCE_POOL_CODE')
            pt = plan_text(content); assert pt['text']  # truncation at the declared cap is allowed and recorded
            plans.append(pt['text'])
            refs.append({'candidate_id': r['candidate_id'], 'sample_idx': int(r['sample_idx']), 'raw_response_path': str(path), 'raw_response_sha256': digest(path),
                         'code_sha256': hashlib.sha256(codes[r['candidate_id']].encode()).hexdigest() if isinstance(codes[r['candidate_id']], str) else None,
                         'plan_sha256': hashlib.sha256(pt['text'].encode()).hexdigest(), 'plan_chars': len(pt['text']),
                         'plan_truncated': pt['truncated'], 'archived_response_had_program': pt['had_program']})
        first = diversity_payload(task, plans)
        control = control_payload(task)
        assert first['messages'][:2] == control['messages']
        tokens = sum(len(enc.encode(m['content'])) + 4 for m in first['messages']) + 3
        control_tokens = sum(len(enc.encode(m['content'])) + 4 for m in control['messages']) + 3
        mean_plan_tokens = sum(len(enc.encode(x)) for x in plans) / 3
        completion = sorted(int(r['completion_tokens']) for r in pool[q])[len(pool[q]) // 2]
        for k in range(SAMPLES):  # step k also carries k earlier diversity plans of about the mean INITIAL plan size
            estimate += Decimal(tokens + k * (mean_plan_tokens + 8)) * Decimal('0.0000025') + Decimal(completion) * Decimal('0.00001')
        estimate += CONTROL_SAMPLES * (Decimal(control_tokens) * Decimal('0.0000025') + Decimal(completion) * Decimal('0.00001'))
        problems.append({'qid': q, 'difficulty': p['difficulty'], 'task_type': p['task_type'], 'archived_S_distinct': p['S_distinct'], 'archived_n_fail': p['n_fail'],
            'public_tests': p['public_tests'], 'initial': refs, 'arm_order': list(arm_order(index)),
            'small_output_space': p['small_output_space'], 'archived_official_passes': archived_passes, 'archived_pool': len(pool[q]),
            'archived_rescue_rate': round(archived_passes / len(pool[q]), 4),
            'step1_payload_sha256': hashlib.sha256(json.dumps(first, sort_keys=True, ensure_ascii=False).encode()).hexdigest(),
            'control_payload_sha256': hashlib.sha256(json.dumps(control, sort_keys=True, ensure_ascii=False).encode()).hexdigest(),
            'step1_prompt_tokens': tokens, 'control_prompt_tokens': control_tokens, 'archived_median_completion_tokens': completion})
    campaign = json.loads(CAMPAIGN_PRIOR.read_text())
    assert campaign['outstanding_requests'] == 0 and Decimal(campaign['remaining_usd']) >= COST_CAP
    dump(STAGE / 'materialization.json', {'scope': SCOPE, 'status': 'MATERIALIZED_OFFLINE_UNAUTHORIZED', 'created_at_utc': datetime.now(timezone.utc).isoformat(),
        'problems': problems, 'calls_planned': (SAMPLES + CONTROL_SAMPLES) * len(problems),
        'initial_plans_truncated': sum(ref['plan_truncated'] for p in problems for ref in p['initial']),
        'initial_responses_without_program': sum(not ref['archived_response_had_program'] for p in problems for ref in p['initial']), 'estimated_cost_usd': str(estimate.quantize(Decimal('0.0001'))),
        'call_cap': CALL_CAP, 'cost_cap_usd': str(COST_CAP), 'prior_campaign': {k: campaign[k] for k in ('provider_calls_total', 'spent_usd', 'remaining_usd')},
        'provider_calls': 0, 'credential_reads': 0, 'hidden_reads': 0, 'candidate_executions': 0})
    dump(STAGE / (P + 'authorization.json'), {'authorized': False, 'user_authorization_confirmed': False, 'scope': SCOPE, 'hard_cap_calls': CALL_CAP,
        'hard_cap_usd': str(COST_CAP), 'current_user_execution_request': None})
    print(json.dumps({'problems': len(problems), 'calls_planned': (SAMPLES + CONTROL_SAMPLES) * len(problems), 'estimated_cost_usd': str(estimate.quantize(Decimal('0.0001')))}))
