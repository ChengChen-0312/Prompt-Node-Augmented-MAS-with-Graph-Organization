def pct(a, b):
    return '%d/%d (%.1f%%)' % (a, b, 100.0 * a / b) if b else '0/0 (—)'


def mcnemar_exact(b, c):
    n = b + c
    if n == 0:
        return 1.0
    return min(1.0, 2 * sum(comb(n, k) for k in range(0, min(b, c) + 1)) / 2 ** n)


def load_scores(run):
    """The whole-batch strict result if it exists, otherwise the per-dataset staged results merged."""
    if (run / 'final_scores.json').exists():
        return read(run / 'final_scores.json')
    parts = [read(run / 'strict' / ('final_scores_%s.json' % d)) for d in DATASETS
             if (run / 'strict' / ('final_scores_%s.json' % d)).exists()]
    if not parts:
        return None
    return {'scope': 'STAGED_DATASETS', 'rows': [r for part in parts for r in part['rows']],
            'summary': {d: v for part in parts for d, v in part['summary'].items()},
            'datasets': {d: v for part in parts for d, v in part['datasets'].items()}}


def load_seals(run):
    if (run / 'generation_seal.json').exists():
        return [read(run / 'generation_seal.json')]
    return [read(p) for p in sorted((run / 'seals').glob('*.json'))] if (run / 'seals').exists() else []


def collect(root):
    """Everything the tables need, per dataset, from saved public artefacts only."""
    root = Path(root)
    run = root / 'runs' / RUN_NAME
    scores = load_scores(run)
    strict = {(r['dataset'], r['question_id'], r['label']): r['strict'] for r in scores['rows']} if scores else {}
    data = {}
    for dataset in DATASETS:
        tasks = load_tasks(dataset, root)
        rows, requests = [], []
        for index, task in enumerate(tasks):
            path = run / 'tasks' / dataset / qdir(index, task['question_id']) / 'task.json'
            state = read(path) if path.exists() else None
            row = {'dataset': dataset, 'question_id': task['question_id'], 'index': index,
                   'task_status': state['status'] if state else 'NOT_STARTED', 'state': state, 'task': task}
            if state:
                for cid, c in state['candidates'].items():
                    requests.append({**c, 'index': index})
            if state and state['status'] == 'COMPLETE':
                d = state['delivery']
                for label in LABELS:
                    c = state['candidates'][d['selected'][label]]
                    s = strict.get((dataset, task['question_id'], label))
                    row[label] = {'candidate_id': c['candidate_id'], 'slot': '%s/%d' % (c['arm'], c['step']),
                                  'public': c['public']['status'], 'code_sha256': c['code_sha256'],
                                  'strict': s['status'] if s else None, 'strict_reason': s.get('reason') if s else None,
                                  'request_dir': c['request_dir'], 'plan': c['plan']}
            rows.append(row)
        data[dataset] = {'rows': rows, 'requests': requests, 'n': len(tasks)}
    return data, scores


def dataset_stats(dataset, entry, scored):
    rows = entry['rows']
    complete = [r for r in rows if r['task_status'] == 'COMPLETE']
    gate = [r for r in complete if r['state']['delivery']['gate_open']]
    s = {'dataset': dataset, 'n': entry['n'], 'started': sum(r['task_status'] != 'NOT_STARTED' for r in rows),
         'initials_complete': sum(1 for r in rows if r['state'] and sum(
             1 for c in r['state']['candidates'].values() if c['arm'] == 'initial') == 3),
         'complete': len(complete), 'd': len(gate), 'not_tested': sum(not r['state']['delivery']['public_tested'] for r in complete),
         'generation_status': 'COMPLETE' if len(complete) == entry['n'] else 'NOT_STARTED' if not any(r['state'] for r in rows) else 'PARTIAL',
         'strict_available': bool(scored and complete and len(complete) == entry['n'] and
                                  all(r[l]['strict'] in ('PASS', 'FAIL') for r in complete for l in LABELS))}
    s['public'] = {l: sum(r[l]['public'] == 'PASS' for r in complete) for l in LABELS}
    s['initial_draft_public_pass'] = [sum(r['state']['delivery']['initial_public'][i] == 'PASS' for r in complete) for i in range(3)]
    s['any_initial_public_pass'] = sum('PASS' in r['state']['delivery']['initial_public'] for r in complete)
    requests = entry['requests']
    def usage(selected):
        return {'requests': len(selected), 'prompt_tokens': sum(c['usage']['prompt_tokens'] for c in selected),
                'completion_tokens': sum(c['usage']['completion_tokens'] for c in selected),
                'seconds': round(sum((c.get('timing') or {}).get('total_seconds', 0) for c in selected), 1)}
    by_arm = {arm: [c for c in requests if c['arm'] == arm] for arm in ('initial', 'a1', 'plan')}
    s['usage'] = {'actual': usage(requests), 'initial': usage(by_arm['initial']),
                  'a1_extra': usage(by_arm['a1']), 'plan_extra': usage(by_arm['plan']),
                  'a1_attributed': usage(by_arm['initial'] + by_arm['a1']),
                  'plan_attributed': usage(by_arm['initial'] + by_arm['plan'])}
    s['format'] = {arm: {'requests': len(cs), 'no_code': sum(c['extraction'] == 'NO_CODE' for c in cs),
                         'length': sum(c['finish_reason'] == 'length' for c in cs),
                         'whole_reply_code': sum(c['extraction'] == 'WHOLE_REPLY_PARSES' for c in cs),
                         'empty_plan': sum(c['plan_is_empty_placeholder'] for c in cs),
                         'plan_truncated': sum(c['plan_truncated'] for c in cs)} for arm, cs in by_arm.items()}
    history = [c for c in by_arm['plan']]
    s['plan_history'] = {'counts': sorted({c['history_plan_count'] for c in history}),
                         'mean_prompt_tokens_plan_extra': round(sum(c['usage']['prompt_tokens'] for c in history) / len(history), 1) if history else None,
                         'mean_prompt_tokens_a1_extra': round(sum(c['usage']['prompt_tokens'] for c in by_arm['a1']) / len(by_arm['a1']), 1) if by_arm['a1'] else None}
    duplicates = {}
    for arm in ('a1', 'plan'):
        dup = total = 0
        for r in gate:
            cands = r['state']['candidates']
            seen = {c['code_sha256'] for c in cands.values() if c['arm'] == 'initial' and c['code_sha256']}
            for step in protocol.STEPS:
                c = cands[protocol.candidate_id(dataset, r['question_id'], arm, step)]
                if c['code_sha256']:
                    total += 1
                    dup += c['code_sha256'] in seen
                    seen.add(c['code_sha256'])
        duplicates[arm] = {'extras_with_code': total, 'identical_to_earlier_visible_candidate': dup}
    s['duplicates'] = duplicates
    s['selected_extra'] = {arm: sum(r[arm]['slot'].startswith(arm) for r in gate) for arm in ('a1', 'plan')}
    if s['strict_available']:
        ok = lambda r, l: r[l]['strict'] == 'PASS'
        s['strict'] = {l: sum(ok(r, l) for r in complete) for l in LABELS}
        s['strict_d'] = {l: sum(ok(r, l) for r in gate) for l in LABELS}
        s['vs_shared'] = {arm: {'rescued': sum(ok(r, arm) and not ok(r, 'shared_initial') for r in complete),
                                'regressed': sum(ok(r, 'shared_initial') and not ok(r, arm) for r in complete)}
                          for arm in ('a1', 'plan')}
        for arm in ('a1', 'plan'):
            s['vs_shared'][arm]['net'] = s['vs_shared'][arm]['rescued'] - s['vs_shared'][arm]['regressed']
        cells = {'both': 0, 'plan_only': 0, 'a1_only': 0, 'neither': 0}
        for r in complete:
            cells['both' if ok(r, 'a1') and ok(r, 'plan') else 'plan_only' if ok(r, 'plan') else 'a1_only' if ok(r, 'a1') else 'neither'] += 1
        s['pair'] = cells
        s['mcnemar_p'] = mcnemar_exact(cells['plan_only'], cells['a1_only'])
        s['public_pass_strict_fail'] = {l: sum(r[l]['public'] == 'PASS' and not ok(r, l) for r in complete) for l in LABELS}
        s['public_fail_strict_pass'] = {l: sum(r[l]['public'] != 'PASS' and ok(r, l) for r in complete) for l in LABELS}
    return s


def static_observation(root, dataset, entry):
    """Cheap textual observations on delivered code; descriptive only, not a correctness judgement."""
    out = {}
    for label in LABELS:
        delivered = [(r, r[label]) for r in entry['rows'] if r['task_status'] == 'COMPLETE']
        no_code = sum(1 for _, d in delivered if not d['code_sha256'])
        flagged = 0
        for r, d in delivered:
            if not d['code_sha256']:
                continue
            code = (Path(root) / d['request_dir'] / 'code.py').read_bytes().decode('utf-8')
            task = r['task']
            if dataset == 'lcb' and task['task_type'] == 'function':
                flagged += not ('class Solution' in code and ('def ' + task['callable']['method']) in code)
            elif dataset == 'usaco' and task['io_mode'] == 'files':
                flagged += not (task['input_filename'] in code and task['output_filename'] in code)
            else:
                flagged += not any(k in code for k in ('input(', 'stdin', 'open(0'))
        out[label] = {'delivered': len(delivered), 'no_code': no_code, 'interface_text_not_found': flagged}
    return out


def write_csvs(root, data):
    reports = Path(root) / 'reports'
    reports.mkdir(exist_ok=True)
    order = 0
    with (reports / '01_逐题配对结果.csv').open('w', encoding='utf-8-sig', newline='') as stream:
        w = csv.writer(stream)
        w.writerow(['order', 'dataset', 'question_id', 'source_index', 'difficulty', 'task_status', 'gate_open_D',
                    'public_tested', 'arm_order', 'initial_public', 'single_initial1_candidate'] +
                   ['%s_%s' % (l, f) for l in LABELS for f in ('slot', 'public', 'strict', 'strict_reason', 'code_sha256')] +
                   ['pair_cell_strict', 'a1_vs_shared', 'plan_vs_shared'])
        for dataset in DATASETS:
            for r in data[dataset]['rows']:
                order += 1
                base = [order, dataset, r['question_id'], r['index'], r['state']['difficulty'] if r['state'] else '', r['task_status']]
                if r['task_status'] != 'COMPLETE':
                    w.writerow(base + [''] * 23)
                    continue
                d = r['state']['delivery']
                cells = []
                for l in LABELS:
                    cells += [r[l]['slot'], r[l]['public'], r[l]['strict'] or '未评分', r[l]['strict_reason'] or '', r[l]['code_sha256'] or 'NO_CODE']
                if all(r[l]['strict'] for l in LABELS):
                    a, p, s0 = (r[l]['strict'] == 'PASS' for l in ('a1', 'plan', 'shared_initial'))
                    pair = 'both' if a and p else 'plan_only' if p else 'a1_only' if a else 'neither'
                    rel = lambda x: 'rescued' if x and not s0 else 'regressed' if s0 and not x else 'same_pass' if x else 'same_fail'
                    tail = [pair, rel(a), rel(p)]
                else:
                    tail = ['未评分'] * 3
                w.writerow(base + [d['gate_open'], d['public_tested'], '>'.join(d['arm_order'] or []),
                                   '|'.join(d['initial_public']), d['selected']['single']] + cells + tail)
    with (reports / '02_请求与资源归属.csv').open('w', encoding='utf-8-sig', newline='') as stream:
        w = csv.writer(stream)
        w.writerow(['dataset', 'question_id', 'source_index', 'candidate_id', 'arm', 'step', 'attributed_to', 'seed',
                    'max_tokens', 'finish_reason', 'prompt_tokens', 'completion_tokens', 'prefill_seconds',
                    'generation_seconds', 'total_seconds', 'generation_tps', 'peak_memory_gb', 'message_count',
                    'history_plan_count', 'plan_chars', 'plan_truncated', 'plan_empty_placeholder', 'extraction',
                    'code_sha256', 'public_status', 'public_pass', 'public_total', 'request_sha256', 'response_sha256', 'request_dir'])
        for dataset in DATASETS:
            for c in data[dataset]['requests']:
                t = c.get('timing') or {}
                pub = c['public'] or {}
                w.writerow([dataset, c['question_id'], c['index'], c['candidate_id'], c['arm'], c['step'],
                            'shared(A1+Plan)' if c['arm'] == 'initial' else protocol.ARM_NAMES[c['arm']], c['seed'],
                            c['max_tokens'], c['finish_reason'], c['usage']['prompt_tokens'], c['usage']['completion_tokens'],
                            t.get('prefill_seconds'), t.get('generation_seconds'), t.get('total_seconds'), t.get('generation_tps'),
                            c.get('peak_memory_gb'), c['message_count'], c['history_plan_count'], c['plan_chars'],
                            c['plan_truncated'], c['plan_is_empty_placeholder'], c['extraction'], c['code_sha256'] or '',
                            pub.get('status', '未公测'), (pub.get('counts') or {}).get('pass', ''), pub.get('total', ''),
                            c['request_sha256'], c['response_sha256'], c['request_dir']])


def technical(root):
    run = Path(root) / 'runs' / RUN_NAME
    states = {}
    for path in sorted((run / 'requests').rglob('state.json')) if (run / 'requests').exists() else []:
        state = read(path)
        if state['state'] != 'COMPLETED':
            states[str(path.parent.relative_to(run))] = {k: state.get(k) for k in ('state', 'logical_id', 'failure_detail', 'partial_stream_chars')}
    unknown = []
    for path in sorted((run / 'tasks').rglob('task.json')) if (run / 'tasks').exists() else []:
        unknown += read(path).get('public_unknown_attempts', [])
    sessions = [json.loads(l) for l in (run / 'logs/sessions.jsonl').read_text(encoding='utf-8').splitlines()] if (run / 'logs/sessions.jsonl').exists() else []
    return {'non_completed_requests': states, 'public_unknown_attempts': unknown,
            'sessions': [{k: s.get(k) for k in ('status', 'reason', 'started_at', 'ended_at')} for s in sessions]}

def verify(root, data, stats, scores, tech):
    from local_gateway import process_alive
    run = Path(root) / 'runs' / RUN_NAME
    seals = load_seals(run)
    seal = seals[0] if seals and seals[0].get('scope') == 'GLOBAL' else None
    pid_file = run / 'worker.pid.json'
    grid = {(r['dataset'], r['question_id'], r['label']) for r in scores['rows']} if scores else set()
    expected = {(d, t['question_id'], l) for d in DATASETS for t in load_tasks(d, root) for l in LABELS}
    summary_ok = bool(scores) and all(
        scores['summary'].get(d, {}).get(l, {}).get('correct') == stats[d].get('strict', {}).get(l) for d in DATASETS for l in LABELS)
    shared_ok = all(len({r['state']['delivery']['selected'][l] for l in ('a1', 'plan')} |
                        {r['state']['delivery']['selected']['shared_initial']}) == 1
                    for d in DATASETS for r in data[d]['rows'] if r['task_status'] == 'COMPLETE' and not r['state']['delivery']['gate_open'])
    seal_ok = bool(seals) and all((Path(root) / p).is_file() and digest(Path(root) / p) == h
                                  for part in seals for p, h in part['files'].items())
    if seals and scores:
        sealed = {(r['dataset'], r['question_id'], r['label']): r['code_sha256'] for part in seals for r in part['selected']}
        seal_ok = seal_ok and all(sealed.get((r['dataset'], r['question_id'], r['label']), 0) == r['code_sha256'] for r in scores['rows'])
    requests = sum(stats[d]['usage']['actual']['requests'] for d in DATASETS)
    checks = {
        '固定题单全部按协议执行（600 题全部完成，无补题/补抽）': all(stats[d]['generation_status'] == 'COMPLETE' for d in DATASETS),
        '所有生成请求有终态，无未知在途或技术失败槽位': not tech['non_completed_requests'],
        '实际请求数等于 3N + 6D': requests == sum(3 * stats[d]['initials_complete'] + 6 * stats[d]['d'] for d in DATASETS),
        '门关闭的题两臂与共同起点交付同一份初稿': shared_ok,
        '全批生成已封存且封存文件哈希全部一致': bool(seal) and seal_ok,
        '已封存部分的文件与严格评分所用源码哈希一致': seal_ok or not seals,
        '严格结果覆盖 1800 个交付标签': grid == expected and len(expected) == 1800,
        '严格结果无 UNKNOWN': bool(scores) and all(r['strict']['status'] in ('PASS', 'FAIL') for r in scores['rows']),
        '严格汇总与逐题一致': summary_ok,
        '模型进程已停止': not pid_file.exists() or not process_alive(read(pid_file)['pid'], str(CODE / 'qwen_worker.py')),
        '远程 API 调用为 0': not seal or seal.get('remote_api_calls') == 0,
    }
    overall = 'COMPLETE' if all(checks.values()) else 'PARTIAL'
    return {'schema': 'QWEN38_FINAL_VERIFICATION_V1', 'campaign_id': CAMPAIGN_ID, 'verified_at': now(),
            'report_version': REPORT_VERSION, 'overall_status': overall, 'checks': checks,
            'datasets': {d: {k: stats[d][k] for k in ('n', 'complete', 'd', 'generation_status', 'strict_available')} for d in DATASETS},
            'formal_requests_completed': requests,
            'generation_seal_sha256': digest(run / 'generation_seal.json') if seal else None,
            'dataset_seals_sha256': {p.stem: digest(p) for p in sorted((run / 'seals').glob('*.json'))} if (run / 'seals').exists() else {},
            'final_scores_sha256': digest(run / 'final_scores.json') if (run / 'final_scores.json').exists() else None}


def bundle(root):
    """Public evidence for independent review. No weights, virtualenv, private tests or credentials exist in this tree."""
    root = Path(root)
    target = root / 'reports' / ('%s_公开证据包.zip' % CAMPAIGN_ID)
    skip_parts = {'__pycache__', 'withdrawn_60task_draft_not_run', '.DS_Store'}
    with zipfile.ZipFile(target, 'w', zipfile.ZIP_DEFLATED, compresslevel=6) as archive:
        for top in ('README_先读.txt', 'PACKAGE_MANIFEST.json', 'handoff', 'data_public', 'code', 'tests', 'preparation', 'runs', 'reports'):
            base = root / top
            for path in ([base] if base.is_file() else sorted(base.rglob('*'))):
                if path.is_file() and not (set(path.parts) & skip_parts) and path != target and path.suffix != '.zip':
                    archive.write(path, str(path.relative_to(root)))
    return target


def main():
    parser = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    parser.add_argument('--root', default=str(ROOT))

                 '- 只比较原 A2-Plan 与 A1；没有 Soft、F、ReAct、A2-Evidence、PN-Update、reviewer 或第三臂。',
                 '- 初稿由本地 Qwen 新生成且两臂共享；没有复用任何 GPT-4o 旧代码、旧失败池或旧修复名单。',
